   178→function parseTags(t: string | null | undefined): string[] {
   179→  if (!t) return [];
   180→  try {
   181→    const v = JSON.parse(t);
   182→    if (Array.isArray(v)) return v.filter((x) => typeof x === "string");
   183→  } catch {
   184→    /* ignore */
   185→  }
   186→  return [];
   187→}

   235→function useDebouncedValue<T>(value: T, delay: number): T {
   236→  const [debounced, setDebounced] = React.useState(value);
   237→  React.useEffect(() => {
   238→    const t = setTimeout(() => setDebounced(value), delay);
   239→    return () => clearTimeout(t);
   240→  }, [value, delay]);
   241→  return debounced;
   242→}

   244→function copyToClipboard(text: string): Promise<void> {
   245→  if (typeof navigator !== "undefined" && navigator.clipboard) {
   246→    return navigator.clipboard.writeText(text);
   247→  }
   248→  return Promise.reject(new Error("Clipboard API not available"));

   616→/** Safely parse the `decodedContent` JSON string on an evidence item. */
   617→function parseDecoded(item: ApiEvidenceItem): Record<string, unknown> | null {
   618→  if (!item.decodedContent) return null;
   619→  try {
   620→    const parsed = JSON.parse(item.decodedContent);
   621→    if (parsed && typeof parsed === "object" && !Array.isArray(parsed)) {
   622→      return parsed as Record<string, unknown>;
   623→    }
   624→    return null;
   625→  } catch {
   626→    return null;
   627→  }
   628→}

   630→function asString(v: unknown): string | null {
   631→  return typeof v === "string" && v.length > 0 ? v : null;

   634→  return typeof v === "number" && !Number.isNaN(v) ? v : null;

   636→function asBool(v: unknown): boolean | null {
   637→  return typeof v === "boolean" ? v : null;
   638→}
   639→function asStringArray(v: unknown): string[] {
   640→  if (!Array.isArray(v)) return [];
   641→  return v.filter((x): x is string => typeof x === "string");

   724→function AudioDecoded({ data }: { item: ApiEvidenceItem; data: Record<string, unknown> }) {
   725→  const [isPlaying, setIsPlaying] = React.useState(false);
   726→  const [progress, setProgress] = React.useState(0); // 0-100
   727→
   728→  const durationSec = asNumber(data.durationSec) ?? 0;
   729→  const durationLabel = asString(data.durationLabel) ?? "0:00";
   730→  const sampleRate = asNumber(data.sampleRate);
   731→  const channels = asNumber(data.channels);
   732→  const codec = asString(data.codec);
   733→  const bitrate = asString(data.bitrate);
   734→  const transcription = asString(data.transcription);
   735→  const confidence = asNumber(data.transcriptionConfidence);
   736→  const language = asString(data.language);
   737→  const waveform = asStringArray(data.waveform)
   738→    .map((v) => Number(v))
   739→    .filter((v) => !Number.isNaN(v));
   740→
   741→  React.useEffect(() => {
   742→    if (!isPlaying || durationSec <= 0) return;
   743→    const tickMs = 200;
   744→    const incrementPerTick = (tickMs / 1000 / durationSec) * 100;
   745→    const interval = setInterval(() => {
   746→      setProgress((p) => {
   747→        const next = p + incrementPerTick;
   748→        return next >= 100 ? 100 : next;
   749→      });
   750→    }, tickMs);
   751→    return () => clearInterval(interval);
   752→  }, [isPlaying, durationSec]);
   753→
   754→  React.useEffect(() => {
   755→    if (progress >= 100 && isPlaying) {
   756→      setIsPlaying(false);
   757→    }
   758→  }, [progress, isPlaying]);
   759→
   760→  function formatTime(percent: number): string {
   761→    if (durationSec <= 0) return "0:00";
   762→    const sec = Math.floor((percent / 100) * durationSec);
   763→    return `${Math.floor(sec / 60)}:${String(sec % 60).padStart(2, "0")}`;
   764→  }
   765→
   766→  function togglePlay() {
   767→    if (progress >= 100) {
   768→      setProgress(0);
   769→      setIsPlaying(true);
   770→    } else {
   771→      setIsPlaying((p) => !p);
   772→    }
   773→  }
   774→
   775→  const confidenceColor =
   776→    confidence == null
   777→      ? null
   778→      : confidence >= 90
   779→        ? "green"
   780→        : confidence >= 75
   781→          ? "amber"
   782→          : "red";
   783→
   784→  return (
   785→    <div className="space-y-3">
   786→      <div className="rounded-md border border-border/60 bg-muted/20 p-3">
   787→        <DecodedSectionHeader icon={Volume2} label="Audio Player" />
   788→
   789→        <div className="flex items-center gap-3">
   790→          <Button
   791→            variant={isPlaying ? "secondary" : "default"}
   792→            className="size-9 shrink-0 rounded-full p-0"
   793→            onClick={togglePlay}
   794→            aria-label={isPlaying ? "Pause" : "Play"}
   795→          >
   796→            {isPlaying ? (
   797→              <Pause className="size-4" />
   798→            ) : (
   799→              <Play className="size-4 translate-x-px" />
   800→            )}
   801→          </Button>
   802→
   803→          <div className="min-w-0 flex-1">
   804→            {/* Waveform visualization */}
   805→            <div className="flex h-10 items-center gap-px">
   806→              {waveform.length > 0 ? (
   807→                waveform.map((v, i) => {
   808→                  const isActive = (i / waveform.length) * 100 <= progress;
   809→                  const heightPct = Math.max(8, Math.min(100, v));
   810→                  return (
   811→                    <div
   812→                      key={i}
   813→                      className={cn(
   814→                        "flex-1 rounded-sm transition-colors",
   815→                        isActive ? "bg-primary" : "bg-muted-foreground/30"
   816→                      )}
   817→                      style={{ height: `${heightPct}%` }}
   818→                    />
   819→                  );
   820→                })
   821→              ) : (
   822→                <div className="text-[11px] text-muted-foreground">
   823→                  No waveform data
   824→                </div>
   825→              )}
   826→            </div>
   827→
   828→            <div className="mt-1 flex items-center justify-between font-mono-forensic text-[10px] text-muted-foreground">
   829→              <span>{formatTime(progress)}</span>
   830→              <span>{durationLabel}</span>
   831→            </div>
   832→          </div>
   833→        </div>
   834→      </div>
   835→
   836→      <div className="rounded-md border border-border/60 bg-muted/20 p-3">
   837→        <DecodedSectionHeader icon={Volume2} label="Audio Metadata" />
   838→        <div className="divide-y divide-border/40">
   839→          <MetaRow label="Duration" value={durationLabel} mono />
   840→          <MetaRow label="Codec" value={codec} mono />
   841→          <MetaRow
   842→            label="Sample Rate"
   843→            value={sampleRate != null ? `${sampleRate.toLocaleString()} Hz` : null}
   844→            mono
   845→          />
   846→          <MetaRow
   847→            label="Channels"
   848→            value={
   849→              channels === 1
   850→                ? "Mono"
   851→                : channels === 2
   852→                  ? "Stereo"
   853→                  : channels != null
   854→                    ? String(channels)
   855→                    : null
   856→            }
   857→          />
   858→          <MetaRow label="Bitrate" value={bitrate} mono />
   859→          <MetaRow label="Language" value={language} mono />
   860→        </div>
   861→      </div>
   862→
   863→      {transcription && (
   864→        <div className="rounded-md border border-border/60 bg-muted/20 p-3">
   865→          <div className="mb-2 flex items-center justify-between gap-2">
   866→            <DecodedSectionHeader icon={MessageSquare} label="Transcription" />
   867→            {confidence != null && confidenceColor && (
   868→              <Badge
   869→                variant="outline"
   870→                className={cn(
   871→                  "gap-1 px-1.5 py-0 text-[10px] font-medium",
   872→                  COLOR_CLASSES[confidenceColor]?.border,
   873→                  COLOR_CLASSES[confidenceColor]?.text,
   874→                  COLOR_CLASSES[confidenceColor]?.bg
   875→                )}
   876→              >
   877→                <Volume2 className="size-2.5" />
   878→                {confidence}% confidence
   879→              </Badge>
   880→            )}
   881→          </div>
   882→          <div className="text-sm leading-relaxed text-foreground italic">
   883→            “{transcription}”
   884→          </div>
   885→        </div>
   886→      )}
   887→    </div>
   888→  );
   889→}

  1349→function humanizeKey(k: string): string {
  1350→  return k
  1351→    .replace(/([a-z0-9])([A-Z])/g, "$1 $2")
  1352→    .replace(/[_-]+/g, " ")
  1353→    .replace(/^./, (s) => s.toUpperCase())
  1354→    .trim();
  1355→}
  1356→
  1357→function GenericJsonDecoded({ data }: { item: ApiEvidenceItem; data: Record<string, unknown> }) {
  1358→  const entries = Object.entries(data);
  1359→  if (entries.length === 0) return null;
  1360→  return (
  1361→    <div className="rounded-md border border-border/60 bg-muted/20 p-3">
  1362→      <DecodedSectionHeader icon={FileText} label="Decoded Content" />
  1363→      <div className="divide-y divide-border/40">
  1364→        {entries.map(([k, v]) => {
  1365→          let value: React.ReactNode;
  1366→          let mono = false;
  1367→          if (v == null) {
  1368→            value = null;
  1369→          } else if (typeof v === "string") {
  1370→            value = v;
  1371→            mono = true;
  1372→          } else if (typeof v === "number" || typeof v === "boolean") {
  1373→            value = String(v);
  1374→            mono = true;
  1375→          } else if (Array.isArray(v)) {
  1376→            value = v.length === 0 ? "[]" : v.map((x) => String(x)).join(", ");
  1377→            mono = true;
  1378→          } else {
  1379→            value = JSON.stringify(v);
  1380→            mono = true;
  1381→          }
  1382→          return (
  1383→            <MetaRow
  1384→              key={k}
  1385→              label={humanizeKey(k)}
  1386→              value={value}
  1387→              mono={mono}
  1388→            />
  1389→          );
  1390→        })}
  1391→      </div>
  1392→    </div>
  1393→  );
  1394→}
