   178→   178→function parseTags(t: string | null | undefined): string[] {
   179→   179→  if (!t) return [];
   180→   180→  try {
   181→   181→    const v = JSON.parse(t);
   182→   182→    if (Array.isArray(v)) return v.filter((x) => typeof x === "string");
   183→   183→  } catch {
   184→   184→    /* ignore */
   185→   185→  }
   186→   186→  return [];
   187→   187→}

   235→   235→function useDebouncedValue<T>(value: T, delay: number): T {
   236→   236→  const [debounced, setDebounced] = React.useState(value);
   237→   237→  React.useEffect(() => {
   238→   238→    const t = setTimeout(() => setDebounced(value), delay);
   239→   239→    return () => clearTimeout(t);
   240→   240→  }, [value, delay]);
   241→   241→  return debounced;
   242→   242→}

   616→   616→/** Safely parse the `decodedContent` JSON string on an evidence item. */
   617→   617→function parseDecoded(item: ApiEvidenceItem): Record<string, unknown> | null {
   618→   618→  if (!item.decodedContent) return null;
   619→   619→  try {
   620→   620→    const parsed = JSON.parse(item.decodedContent);
   621→   621→    if (parsed && typeof parsed === "object" && !Array.isArray(parsed)) {
   622→   622→      return parsed as Record<string, unknown>;
   623→   623→    }
   624→   624→    return null;
   625→   625→  } catch {
   626→   626→    return null;
   627→   627→  }
   628→   628→}

   630→   630→function asString(v: unknown): string | null {
   631→   631→  return typeof v === "string" && v.length > 0 ? v : null;
   632→   632→}
   633→   633→function asNumber(v: unknown): number | null {
   634→   634→  return typeof v === "number" && !Number.isNaN(v) ? v : null;
   635→   635→}
   636→   636→function asBool(v: unknown): boolean | null {
   637→   637→  return typeof v === "boolean" ? v : null;
   638→   638→}
   639→   639→function asStringArray(v: unknown): string[] {
   640→   640→  if (!Array.isArray(v)) return [];
   641→   641→  return v.filter((x): x is string => typeof x === "string");
   642→   642→}

   723→   723→
   724→   724→function AudioDecoded({ data }: { item: ApiEvidenceItem; data: Record<string, unknown> }) {
   725→   725→  const [isPlaying, setIsPlaying] = React.useState(false);
   726→   726→  const [progress, setProgress] = React.useState(0); // 0-100
   727→   727→
   728→   728→  const durationSec = asNumber(data.durationSec) ?? 0;
   729→   729→  const durationLabel = asString(data.durationLabel) ?? "0:00";
   730→   730→  const sampleRate = asNumber(data.sampleRate);
   731→   731→  const channels = asNumber(data.channels);
   732→   732→  const codec = asString(data.codec);
   733→   733→  const bitrate = asString(data.bitrate);
   734→   734→  const transcription = asString(data.transcription);
   735→   735→  const confidence = asNumber(data.transcriptionConfidence);
   736→   736→  const language = asString(data.language);
   737→   737→  const waveform = asStringArray(data.waveform)
   738→   738→    .map((v) => Number(v))
   739→   739→    .filter((v) => !Number.isNaN(v));
   740→   740→
   741→   741→  React.useEffect(() => {
   742→   742→    if (!isPlaying || durationSec <= 0) return;
   743→   743→    const tickMs = 200;
   744→   744→    const incrementPerTick = (tickMs / 1000 / durationSec) * 100;
   745→   745→    const interval = setInterval(() => {
   746→   746→      setProgress((p) => {
   747→   747→        const next = p + incrementPerTick;
   748→   748→        return next >= 100 ? 100 : next;
   749→   749→      });
   750→   750→    }, tickMs);
   751→   751→    return () => clearInterval(interval);
   752→   752→  }, [isPlaying, durationSec]);
   753→   753→
   754→   754→  React.useEffect(() => {
   755→   755→    if (progress >= 100 && isPlaying) {
   756→   756→      setIsPlaying(false);
   757→   757→    }
   758→   758→  }, [progress, isPlaying]);
   759→   759→
   760→   760→  function formatTime(percent: number): string {
   761→   761→    if (durationSec <= 0) return "0:00";
   762→   762→    const sec = Math.floor((percent / 100) * durationSec);
   763→   763→    return `${Math.floor(sec / 60)}:${String(sec % 60).padStart(2, "0")}`;
   764→   764→  }
   765→   765→
   766→   766→  function togglePlay() {
   767→   767→    if (progress >= 100) {
   768→   768→      setProgress(0);
   769→   769→      setIsPlaying(true);
   770→   770→    } else {
   771→   771→      setIsPlaying((p) => !p);
   772→   772→    }
   773→   773→  }
   774→   774→
   775→   775→  const confidenceColor =
   776→   776→    confidence == null
   777→   777→      ? null
   778→   778→      : confidence >= 90
   779→   779→        ? "green"
   780→   780→        : confidence >= 75
   781→   781→          ? "amber"
   782→   782→          : "red";
   783→   783→
   784→   784→  return (
   785→   785→    <div className="space-y-3">
   786→   786→      <div className="rounded-md border border-border/60 bg-muted/20 p-3">
   787→   787→        <DecodedSectionHeader icon={Volume2} label="Audio Player" />
   788→   788→
   789→   789→        <div className="flex items-center gap-3">
   790→   790→          <Button
   791→   791→            variant={isPlaying ? "secondary" : "default"}
   792→   792→            className="size-9 shrink-0 rounded-full p-0"
   793→   793→            onClick={togglePlay}
   794→   794→            aria-label={isPlaying ? "Pause" : "Play"}
   795→   795→          >
   796→   796→            {isPlaying ? (
   797→   797→              <Pause className="size-4" />
   798→   798→            ) : (
   799→   799→              <Play className="size-4 translate-x-px" />
   800→   800→            )}
   801→   801→          </Button>
   802→   802→
   803→   803→          <div className="min-w-0 flex-1">
   804→   804→            {/* Waveform visualization */}
   805→   805→            <div className="flex h-10 items-center gap-px">
   806→   806→              {waveform.length > 0 ? (
   807→   807→                waveform.map((v, i) => {
   808→   808→                  const isActive = (i / waveform.length) * 100 <= progress;
   809→   809→                  const heightPct = Math.max(8, Math.min(100, v));
   810→   810→                  return (
   811→   811→                    <div
   812→   812→                      key={i}
   813→   813→                      className={cn(
   814→   814→                        "flex-1 rounded-sm transition-colors",
   815→   815→                        isActive ? "bg-primary" : "bg-muted-foreground/30"
   816→   816→                      )}
   817→   817→                      style={{ height: `${heightPct}%` }}
   818→   818→                    />
   819→   819→                  );
   820→   820→                })
   821→   821→              ) : (
   822→   822→                <div className="text-[11px] text-muted-foreground">
   823→   823→                  No waveform data
   824→   824→                </div>
   825→   825→              )}
   826→   826→            </div>
   827→   827→
   828→   828→            <div className="mt-1 flex items-center justify-between font-mono-forensic text-[10px] text-muted-foreground">
   829→   829→              <span>{formatTime(progress)}</span>
   830→   830→              <span>{durationLabel}</span>
   831→   831→            </div>
   832→   832→          </div>
   833→   833→        </div>
   834→   834→      </div>
   835→   835→
   836→   836→      <div className="rounded-md border border-border/60 bg-muted/20 p-3">
   837→   837→        <DecodedSectionHeader icon={Volume2} label="Audio Metadata" />
   838→   838→        <div className="divide-y divide-border/40">
   839→   839→          <MetaRow label="Duration" value={durationLabel} mono />
   840→   840→          <MetaRow label="Codec" value={codec} mono />
   841→   841→          <MetaRow
   842→   842→            label="Sample Rate"
   843→   843→            value={sampleRate != null ? `${sampleRate.toLocaleString()} Hz` : null}
   844→   844→            mono
   845→   845→          />
   846→   846→          <MetaRow
   847→   847→            label="Channels"
   848→   848→            value={
   849→   849→              channels === 1
   850→   850→                ? "Mono"
   851→   851→                : channels === 2
   852→   852→                  ? "Stereo"
   853→   853→                  : channels != null
   854→   854→                    ? String(channels)
   855→   855→                    : null
   856→   856→            }
   857→   857→          />
   858→   858→          <MetaRow label="Bitrate" value={bitrate} mono />
   859→   859→          <MetaRow label="Language" value={language} mono />
   860→   860→        </div>
   861→   861→      </div>
   862→   862→
   863→   863→      {transcription && (
   864→   864→        <div className="rounded-md border border-border/60 bg-muted/20 p-3">
   865→   865→          <div className="mb-2 flex items-center justify-between gap-2">
   866→   866→            <DecodedSectionHeader icon={MessageSquare} label="Transcription" />
   867→   867→            {confidence != null && confidenceColor && (
   868→   868→              <Badge
   869→   869→                variant="outline"
   870→   870→                className={cn(
   871→   871→                  "gap-1 px-1.5 py-0 text-[10px] font-medium",
   872→   872→                  COLOR_CLASSES[confidenceColor]?.border,
   873→   873→                  COLOR_CLASSES[confidenceColor]?.text,
   874→   874→                  COLOR_CLASSES[confidenceColor]?.bg
   875→   875→                )}
   876→   876→              >
   877→   877→                <Volume2 className="size-2.5" />
   878→   878→                {confidence}% confidence
   879→   879→              </Badge>
   880→   880→            )}
   881→   881→          </div>
   882→   882→          <div className="text-sm leading-relaxed text-foreground italic">
   883→   883→            “{transcription}”
   884→   884→          </div>
   885→   885→        </div>
   886→   886→      )}
   887→   887→    </div>
   888→   888→  );
   889→   889→}
