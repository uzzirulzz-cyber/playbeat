   165→function parseTags(t: string | null | undefined): string[] {
   166→  if (!t) return [];
   167→  try {
   168→    const v = JSON.parse(t);
   169→    if (Array.isArray(v)) return v.filter((x) => typeof x === "string");
   170→  } catch {
   171→    /* ignore */
   172→  }
   173→  return [];
   174→}

   177→  if (c >= 80) return "high";

   222→function useDebouncedValue<T>(value: T, delay: number): T {
   223→  const [debounced, setDebounced] = React.useState(value);
   224→  React.useEffect(() => {
   225→    const t = setTimeout(() => setDebounced(value), delay);
   226→    return () => clearTimeout(t);
   227→  }, [value, delay]);
   228→  return debounced;
   229→}

   231→function copyToClipboard(text: string): Promise<void> {
   232→  if (typeof navigator !== "undefined" && navigator.clipboard) {
   233→    return navigator.clipboard.writeText(text);
   234→  }
   235→  return Promise.reject(new Error("Clipboard API not available"));
